    Scripting Element
    a) Comment <%-- comment --%>
    b) Directive - they direction to server regarding the page
               - <%@ directive %>
               example
               <%@ page import="" %>
    c) Declaration <%! declarations %>
                   <%! int loginTime = new Date() %>

    d) Scriptlets <% code %>

    e) Expression - display as string
            <%= loginTime %>

    Implicit Objects
    a) out = JspWriter
        - out.print();
        - out.println();
        - out.newLine();
    b) request = HttpServletRequest
    request.getParameter("name")         // single value from form/URL: ?name=Brian
request.getParameterValues("hobby")  // multiple values: ?hobby=coding&hobby=music
request.getParameterNames()          // all param names as Enumeration
request.getParameterMap()            // all params as a Map<String, String[]>
request.getHeader("User-Agent")      // browser info, referrer, etc.
request.setAttribute("key", value)   // attach data to the request (for forwarding)
request.getAttribute("key")          // retrieve it later
        
    c) response = HttpServletResponse
        - response.setContentType(String type)
        - response.setHeader(String name, String value)
        - response.getWriter()
    d) session = HttpSession
    e) application = ServletContext
    f) config = ServletConfig
